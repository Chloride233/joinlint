# JoinLint v0 Implementation Plan

> **For agentic workers:** Execute this plan in the current session by default, task by task. Use checkbox (`- [ ]`) syntax for tracking. Use subagents only when the user explicitly requests delegation or parallel agent work.

**Goal:** Ship a local, deterministic CLI and STDIO MCP service that discovers, verifies, and continuously checks safe single-column joins across CSV directories and SQLite databases.

**Architecture:** A typed Python core owns contracts, trusted project paths, snapshots, exact profiling, candidate scoring, validation, and atomic generated artifacts. Typer CLI and MCP are deliberately thin adapters over that core; neither accepts SQL nor mutates data sources. Git-tracked YAML and baselines contain human decisions and sanitized evidence, while generated scan artifacts and candidate rejections remain local.

**Tech Stack:** Python 3.12+, Typer, PyYAML, Pydantic, `rfc8785`, stdlib `sqlite3`/`csv`, MCP Python SDK, pytest, Hypothesis, Ruff, GitHub Actions.

## Global Constraints

- Support macOS and Linux only; reject operation when descriptor-relative no-follow path handling is unavailable.
- Support only local CSV directories and SQLite files; do not add Parquet, DuckDB, remote databases, network access, LLM calls, or source writes.
- Scan every registered source as one atomic transaction; failures preserve the last generated transaction and return `inconclusive`.
- Treat every safety claim as exact evidence; a limit or mutation failure is never a pass.
- Enforce each configured `max_scan_seconds` with a monotonic source deadline during CSV copying, rewalks, SQLite backups, and backup progress callbacks; expiry is `INCONCLUSIVE_SCAN` and publishes no evidence.
- Keep confirmed `model.yaml` decisions in Git; generated artifacts and `state/rejections.json` remain ignored local state.
- Never return raw rows, credentials, absolute paths, arbitrary SQL results, or unrestricted query execution through CLI JSON, reports, or MCP.
- Use schema version `1`, UTF-8-byte deterministic ordering, a maximum 1 MiB JSON/MCP response, and exit codes `0`, `1`, `2`, `3`, `4` exactly as specified.
- Record scanner, policy, source, model, and relationship digests in generated evidence; use RFC 8785 canonical JSON for model and rejection-evidence digests.
- Keep all code changes covered by focused pytest tests; run the full suite, Ruff, `git diff --check`, and repository status before each stage handoff.

```python
from typing import Any, Literal
import rfc8785
from pydantic import BaseModel

Status = Literal["ok", "findings", "inconclusive", "error"]

class ErrorBody(BaseModel):
    code: str
    message: str

class Finding(BaseModel):
    code: str
    severity: Literal["info", "warning", "blocking"]
    message: str

class Envelope(BaseModel):
    schema_version: Literal[1] = 1
    command: str
    status: Status
    data: dict[str, Any] | None
    findings: list[Finding] = []
    error: ErrorBody | None = None

def canonical_json(value: Any) -> bytes:
    return rfc8785.dumps(value)

def envelope_for(command: str, status: Status, error_code: str | None = None) -> Envelope:
    error = ErrorBody(code=error_code, message=error_code) if error_code else None
    return Envelope(command=command, status=status, data=None if error else {}, error=error)
```

```python
class SafeProject:
    def __init__(self, root: Path) -> None:
        if os.name != "posix":
            raise JoinLintError("PLATFORM_UNSUPPORTED", "macOS or Linux is required", 2)
        self.root = root.resolve(strict=True)
        self._root_fd = os.open(self.root, os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC)

    def open_relative(self, relative: PurePosixPath, flags: int) -> int:
        if relative.is_absolute() or ".." in relative.parts or not relative.parts:
            raise JoinLintError("PATH_OUTSIDE_PROJECT", "source path is not project-relative", 2)
        fd = os.dup(self._root_fd)
        try:
            for index, part in enumerate(relative.parts):
                open_flags = flags if index == len(relative.parts) - 1 else os.O_RDONLY | os.O_DIRECTORY
                next_fd = os.open(part, open_flags | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=fd)
                os.close(fd)
                fd = next_fd
            return fd
        except OSError as exc:
            os.close(fd)
            raise JoinLintError("SAFE_PATH_OPEN_FAILED", str(exc), 2) from exc
```

```python
def fingerprint(entries: list[SnapshotFile]) -> str:
    digest = hashlib.sha256(b"joinlint-source-v1")
    for entry in sorted(entries, key=lambda item: os.fsencode(item.relative_path.as_posix())):
        encoded = entry.relative_path.as_posix().encode("utf-8")
        digest.update(len(encoded).to_bytes(8, "big"))
        digest.update(encoded)
        digest.update(entry.size.to_bytes(8, "big"))
        digest.update(bytes.fromhex(entry.sha256))
    return digest.hexdigest()

def snapshot_sqlite(parent_fd: int, database_name: str, expected: os.stat_result,
                    destination: Path, deadline: Deadline) -> None:
    uri = f"file:/dev/fd/{parent_fd}/{quote(database_name, safe='')}?mode=ro"
    source = sqlite3.connect(uri, uri=True)
    target = sqlite3.connect(destination)
    try:
        source.execute("PRAGMA query_only = ON")
        source.execute("BEGIN")
        source.backup(target, pages=128, progress=lambda *_: deadline.check())
    finally:
        source.close()
        target.close()
    actual = os.stat(database_name, dir_fd=parent_fd, follow_symlinks=False)
    if (actual.st_dev, actual.st_ino, actual.st_size, actual.st_mtime_ns) != (
        expected.st_dev, expected.st_ino, expected.st_size, expected.st_mtime_ns
    ):
        raise JoinLintError("SOURCE_CHANGED_DURING_SCAN", "source changed during backup", 3)
```

Before this function, the adapter opens the database leaf descriptor-relatively
and records its device, inode, size, and nanosecond modification time. It keeps
the verified parent-directory FD open through the backup, constructs the SQLite
URI only from that parent FD and the validated leaf name, and checks the source
deadline before opening, beginning the transaction, and every backup callback.
It repeats the descriptor-relative identity check after the backup. This
preserves the conventional SQLite `-wal` lookup beside the verified database
while detecting a mutation before any evidence is used. A deadline expiry,
backup failure, or identity mismatch returns a stable exit-3-class error.

```python
@dataclass(frozen=True)
class ColumnProfile:
    name: str
    physical_type: Literal["integer", "number", "text", "boolean", "unknown"]
    null_count: int
    distinct_count: int
    is_unique: bool

def profile_column(name: str, values: Iterable[object]) -> ColumnProfile:
    all_values = list(values)
    non_null = [value for value in all_values if value not in (None, "")]
    return ColumnProfile(
        name=name,
        physical_type=infer_type(non_null),
        null_count=len(all_values) - len(non_null),
        distinct_count=len(set(normalize_scalar(value) for value in non_null)),
        is_unique=len(non_null) == len(set(normalize_scalar(value) for value in non_null)),
```

```python
    evidence = {
        "endpoints": candidate.endpoints,
        "types": candidate.types,
        "source_fingerprints": candidate.source_fingerprints,
        "matched_distinct_count": candidate.evidence.matched_distinct_count,
        "inclusion_numerator": candidate.evidence.inclusion_numerator,
```

```python
def validate_relationship(edge: Relationship, catalog: ScanCatalog) -> ValidationResult:
    child, parent = catalog.endpoint(edge.from_), catalog.endpoint(edge.to)
    findings: list[Finding] = []
    if not parent.is_unique:
        findings.append(blocking("REFERENCED_KEY_NOT_UNIQUE"))
    if child.null_count:
        findings.append(warning("CHILD_KEY_NULL"))
    if child.orphan_count(parent):
        findings.append(blocking("ORPHAN_CHILD_ROW"))
    observed = cardinality(child, parent)
    if observed != edge.cardinality:
        findings.append(blocking("CARDINALITY_DRIFT"))
    return ValidationResult.from_profiles(edge, child, parent, observed, findings)
```

```python
def write_generated_transaction(project: Path, files: dict[str, bytes]) -> None:
    generated = project / ".joinlint" / "generated"
    temporary = Path(tempfile.mkdtemp(prefix="generated-", dir=generated.parent))
    try:
        for name, body in files.items():
            target = temporary / name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(body)
        os.replace(temporary, generated)
    except Exception:
        shutil.rmtree(temporary, ignore_errors=True)
        raise

def render_report(data: ReportData) -> str:
    title = html.escape("JoinLint report", quote=True)
    rows = "".join(f"<li>{html.escape(name, quote=True)}</li>" for name in data.identifiers)
    return f"<!doctype html><meta http-equiv=\"Content-Security-Policy\" content=\"default-src 'none'; style-src 'unsafe-inline'\"><title>{title}</title><ul>{rows}</ul>"
```
